cmake_minimum_required(VERSION 3.16)
project(guidepup_nav2_bridge)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(pluginlib REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Boost REQUIRED)
link_directories("$ENV{NAV2_SYSROOT}/usr/lib/x86_64-linux-gnu" "$ENV{NAV2_SYSROOT}/usr/lib/llvm-18/lib" "$ENV{NAV2_PREFIX}/lib")
add_library(guidepup_envelope SHARED envelope.cpp)
ament_target_dependencies(guidepup_envelope nav2_mppi_controller pluginlib)
target_link_libraries(guidepup_envelope nlohmann_json::nlohmann_json)
target_include_directories(guidepup_envelope PRIVATE ${Boost_INCLUDE_DIRS})
pluginlib_export_plugin_description_file(nav2_mppi_controller plugins.xml)
add_executable(guidepup_mppi_bridge bridge.cpp)
target_compile_definitions(guidepup_mppi_bridge PRIVATE NAV2_MPPI_VERSION="${nav2_mppi_controller_VERSION}")
ament_target_dependencies(guidepup_mppi_bridge nav2_mppi_controller nav2_costmap_2d rclcpp rclcpp_lifecycle)
target_link_libraries(guidepup_mppi_bridge guidepup_envelope nlohmann_json::nlohmann_json)
install(TARGETS guidepup_envelope guidepup_mppi_bridge LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
